cmake_minimum_required(VERSION 3.21)
option(TUFTY_FIRMWARE "Build Tufty2350 standalone firmware" OFF)
if(TUFTY_FIRMWARE)
  include(cmake/pico.cmake)
endif()
project(tufty_infones_plus VERSION 0.1.0 LANGUAGES C CXX)
if(TUFTY_FIRMWARE)
  enable_language(ASM)
  pico_sdk_init()
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(TUFTY_FIRMWARE)
  set(sanitizer_default OFF)
else()
  set(sanitizer_default ON)
endif()
option(NES_SANITIZERS "Address/undefined behavior checks on host" ${sanitizer_default})
if(TUFTY_FIRMWARE AND NES_SANITIZERS)
  message(FATAL_ERROR "Use separate host/device build directories; device sanitizers are unsupported")
endif()
if(NES_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_subdirectory(infones)
add_library(nes_cartridge STATIC core_api/cartridge.cpp adapters/infones/cartridge.cpp)
target_include_directories(nes_cartridge PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nes_cartridge PUBLIC nes_core)
add_library(nes_host STATIC adapters/infones/host.cpp)
target_link_libraries(nes_host PUBLIC nes_cartridge)
include(CTest)
option(TUFTY_BUSINESS_TESTS "Build storage, USB, BLE and application host regressions" OFF)
if(BUILD_TESTING AND NOT TUFTY_FIRMWARE)
  add_subdirectory(tests)
endif()
if(TUFTY_FIRMWARE)
  include(cmake/firmware.cmake)
endif()
